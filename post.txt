<script>
let current = 0;
let audio;
let visible = false;

function show()
{ if(visible) 
  { document.getElementById("display").style.display="none";
    document.getElementById("show").textContent = "Show Conversation";
    visible = false;
  } else
  { document.getElementById("display").style.display="block";
    document.getElementById("show").textContent = "Hide Conversation";
    visible = true;
  }
}

function play()
{ current = 0;
  audio.pause();
  playNext();
}
  
function playNext() {
    if (current >= text.length) return;

    audio = new Audio(text[current].audioUrl);

    audio.onended = () => {
        current++;
        playNext();
    };

    audio.play().catch(error => {
        console.error("Playback failed:", error);
    });
}

document.addEventListener('click', () => {
    playNext();
}, { once: true });

{ let str=""; let styles = [];
  for(let i=0; i<voices.length; i++)

  for(let i=0; i<text.length; i++)
  { str += `<div class="conversation ${styles[text[i].voice]}">
            ${text[i].text}
        </div>`;
  }
  document.getElementById("display").innerHTML = str;
}
</script>
